# Запуск шлюза:  deno run --allow-net --allow-env --allow-read ai.ts

# Порт ИИ-шлюза (чат на 8001 не занимает)
AI_PORT=8002

# Кто первый в очереди роутера: GLM | DeepSeek | VseGPT | OpenRouter | Groq
# Пусто = порядок по умолчанию (GLM → DeepSeek → VseGPT → OpenRouter → Groq)

# --- Ключи. Заполнять только те, что есть. Остальные оставить пустыми. ---

# 2) DeepSeek — платно, копейки, из РФ работает напрямую
DEEPSEEK_KEY=
DEEPSEEK_MODEL=deepseek-chat

# 3) VseGPT (vsegpt.ru) — российский шлюз, есть бесплатные модели
VSEGPT_KEY=
VSEGPT_MODEL=gpt-3.5-turbo

# 4) OpenRouter — бесплатные модели с суффиксом :free
OPENROUTER_KEY=
OPENROUTER_MODEL=meta-llama/llama-3.1-8b-instruct:free

# 5) Groq — очень быстро, free tier
GROQ_KEY=
GROQ_MODEL=llama-3.3-70b-versatile
